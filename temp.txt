  --query SecretString --output text | \
  jq '.KAFKA_ENABLED = "false"' | \

for svc in backend eval pca voice-service; do
  K8S_NAME="convogent-${svc%-service}-secrets"

  SECRET_JSON=$(aws secretsmanager get-secret-value \
    --secret-id "convogent/dev/$svc" \
    --region ap-south-1 \
    --query SecretString --output text)

  kubectl delete secret "$K8S_NAME" -n convogent --ignore-not-found
  kubectl create secret generic "$K8S_NAME" \
    -n convogent \
    --from-env-file=<(echo "$SECRET_JSON" | jq -r 'to_entries[] | "\(.key)=\(.value)"')
done

kubectl rollout restart deployment -n convogent



# Get current secret, update the MongoDB URI, push it back
aws secretsmanager get-secret-value --secret-id convogent/dev/backend --region ap-south-1 \
  --query SecretString --output text | \
  jq '.MONGO_DB_URI = "mongodb://<USERNAME>:<PASSWORD>@<DOCDB_ENDPOINT>:27017/?tls=true&tlsCAFile=global-bundle.pem&replicaSet=rs0&readPreference=secondaryPreferred&retryWrites=false"' | \
  xargs -0 aws secretsmanager put-secret-value --secret-id convogent/dev/backend --region ap-south-1 --secret-string

SECRET_JSON=$(aws secretsmanager get-secret-value \
